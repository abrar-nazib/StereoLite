\section{Recurrent Tile Refinement}
% Source: FINAL_MODEL_ARCHITECTURE section 7. Separate unshared
% TileRefine modules at 1/16, 1/8, 1/4 (channel counts differ); warp +
% local correlation lookup (half range 2, five offsets); the gated
% residual head (four sigmoid gates); softplus non-negativity; schedule
% 2/3/3 = eight updates.
\todo{Write per WRITING\_PLAN section 5.1 row 3.5, using
\eqref{eq:gru_input} and \eqref{eq:convgru}.}

\begin{equation}
x \;=\; \bigl[\, f_{L},\; \mathcal{W}(f_{R};\,d),\; d,\; s_{x},\; s_{y},\;
  c,\; \mathrm{corr},\; \mathrm{ctx} \,\bigr]
\label{eq:gru_input}
\end{equation}
\todo{where-clause: $\mathcal{W}$ the disparity warp of the right
features, $\mathrm{corr}$ the five-offset local correlation, and
$\mathrm{ctx}$ the scale-matched context map.}

\begin{equation}
\begin{aligned}
z &= \sigma\!\bigl(\mathrm{Conv}([h,\,x])\bigr), &
r &= \sigma\!\bigl(\mathrm{Conv}([h,\,x])\bigr),\\
q &= \tanh\!\bigl(\mathrm{Conv}([r \odot h,\,x])\bigr), &
h' &= (1-z) \odot h + z \odot q,
\end{aligned}
\label{eq:convgru}
\end{equation}
\todo{where-clause: $h$ the 32-channel persistent hidden state, $\odot$
the element-wise product.}

\section{Plane-Aware Cross-Scale Propagation}
% Source: FINAL_MODEL_ARCHITECTURE section 8. Slopes are USED, not just
% stored; contrast with plain bilinear upsampling.

\begin{equation}
d_{\mathrm{child}} \;=\; 2\,\mathcal{B}(d_{\mathrm{parent}})
  \;+\; 2\,s_{x}\,\Delta x \;+\; 2\,s_{y}\,\Delta y,
\qquad \Delta x, \Delta y \in \{-0.25,\, +0.25\}
\label{eq:plane_prop}
\end{equation}
\todo{where-clause: $\mathcal{B}$ bilinear resampling; the factor two

\section{Geometry Encoding Volume and Fail-Soft Fusion}
% Source: FINAL_MODEL_ARCHITECTURE section 9 + the narrow-band variant
% (the trained configuration): the volume samples 33 hypotheses in a
% +-16 band around the current tile disparity (efficiency finding F3,
% validated accuracy-neutral). Three 3x3x3 convolutions, 16 channels.
\todo{Write per WRITING\_PLAN section 5.1 row 3.7, around
\eqref{eq:gev_expect} and \eqref{eq:failsoft}; explain the bias-init
argument (initial gate weight ~0.018, training starts at the stable
baseline and must learn where GEV information helps).}

\begin{equation}
d_{\mathrm{gev}} = \sum_{d} p(d)\, d, \qquad
c_{\mathrm{gev}} = \max_{d}\, p(d), \qquad
g_{\mathrm{gev}} = \phi\Bigl(\sum_{d} G(d)\, p(d)\Bigr)
\label{eq:gev_expect}
\end{equation}
\todo{where-clause: $G$ the regularized 3-D geometry feature, $\phi$ a
projection to a 16-channel 2-D expected-geometry map.}

\begin{equation}
\begin{aligned}
w &= \sigma\!\Bigl(F\bigl([\mathrm{ctx}_{4},\, g_{\mathrm{gev}},\, c,\,
  c_{\mathrm{gev}},\, \lvert d_{\mathrm{gev}} - d \rvert,\, d]\bigr)\Bigr),\\
d_{\mathrm{fused}} &= \mathrm{softplus}\bigl(d + w\,(d_{\mathrm{gev}} - d)\bigr),
\end{aligned}
\label{eq:failsoft}
\end{equation}
\todo{where-clause: $F$ the fusion network with final-layer bias
initialized to $-4$; slopes attenuate by $(1-w)$; confidence becomes the
maximum of the two estimates.}

\section{Training Objective}

\begin{equation}
\begin{aligned}
\mathcal{L} \;=\;& 1.00\,\mathcal{L}_{1}(d_{\mathrm{full}})
 + 0.50\,\mathcal{L}_{1}(d_{1/2})
 + 0.30\,\mathcal{L}_{1}(d_{1/4})
 + 0.20\,\mathcal{L}_{1}(d_{1/8})
 + 0.10\,\mathcal{L}_{1}(d_{1/16})\\

 + 0.30\,\mathcal{L}_{\mathrm{slant}}
\end{aligned}

\end{equation}

\section{Evaluation Metrics}
% Source: FINAL_MODEL_ARCHITECTURE section 16.
\todo{Introduce \eqref{eq:metrics}; add RMSE and median absolute error in
prose; state the latency and memory measurement protocol (batch 1,
384x640, warm-up, fp16/fp32, device named per measurement).}

\begin{equation}
\begin{aligned}
\mathrm{EPE} &= \frac{1}{\lvert V \rvert} \sum_{p \in V}
  \bigl\lvert \hat{d}(p) - d^{*}(p) \bigr\rvert, \qquad
\mathrm{bad}\text{-}t = \frac{100}{\lvert V \rvert}\,
  \Bigl\lvert \bigl\{ p \in V :
  \lvert \hat{d}(p) - d^{*}(p) \rvert > t \bigr\} \Bigr\rvert,\\
\mathrm{D1} &= \frac{100}{\lvert V \rvert}\, \Bigl\lvert \bigl\{ p \in V :
  \lvert \hat{d}(p) - d^{*}(p) \rvert > 3 \;\wedge\;
  \lvert \hat{d}(p) - d^{*}(p) \rvert > 0.05\, d^{*}(p) \bigr\} \Bigr\rvert
\end{aligned}
\label{eq:metrics}
\end{equation}
\todo{where-clause: $V$ the valid-pixel set, $\hat{d}$ the prediction,
$d^{*}$ the ground truth, $t \in \{0.5, 1, 2, 3\}$ px.}
